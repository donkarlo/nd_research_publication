    \begin{frame}[allowframebreaks]{Discrepancy detection}
        Discrepancy
        \begin{itemize}
            \item \textbf{Anomaly}: Short-term value above the threshold
            \item \textbf{Novelty}: Continuously high anomaly
            \item \textbf{Outlier}: Statistically distant from the observation distribution but still a valid reading
        \end{itemize}

        \textbf{Anomaly} is the Mahalanobis distance between the prediction and the preprocessed, fused observation.


        \section{Threshold}
            \begin{enumerate}
                \item Breaking event-specific sensory data into periods
                \item Split the periods into training and test sets
                \item Take $\mu + n \times \sigma$ of the anomaly values as the threshold
            \end{enumerate}

            \begin{algorithm}[H]
		\caption{Novelty Detection from Sensor Sequences}
        \begin{algorithmic}[1]
			\Require Preprocessed sensor sequence $X = \{x_1, x_2, \dots, x_T\}$
            \Require Trained predictive model $f_{\theta}$
            \Ensure Novelty threshold $\tau = \{y_1, y_2, \dots, y_T\}$
            \Ensure Novelty scalar value


            $Y = \{y_1, y_2, \dots, y_T\}$
            \State \Return $Y$
        \end{algorithmic}
            \end{algorithm}
