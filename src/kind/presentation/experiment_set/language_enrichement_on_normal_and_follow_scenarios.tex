\begin{frame}[allowframebreaks]{Language testing}
    
    \begin{itemize}
        \item Learning the language of the normal scenario
        \item Learning the language of the follow scenario
        \item Extracting the language components of the follow scenario
        \item Enriching the language from the normal scenario with the missing components from the follow scenario (components = phonemes, words, \.\.\.)
        \item Using the data from the follow scenario, except for the part that differs from the normal scenario (where novelty occurred), and checking whether the discrepancy value remains below the threshold
    \end{itemize}

\end{frame}
